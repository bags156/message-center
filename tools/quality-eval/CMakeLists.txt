cmake_minimum_required(VERSION 3.22)
project(quality_eval LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# QUALITY_ARM_ARCH: build for a fixed ARM level (as the app does) instead of this machine's own.
if(QUALITY_ARM_ARCH)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_CPU_ARM_ARCH ${QUALITY_ARM_ARCH} CACHE STRING "" FORCE)
else()
  set(GGML_NATIVE ON CACHE BOOL "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(root ${CMAKE_CURRENT_SOURCE_DIR}/../../quality-native)
add_subdirectory(${root}/engine llama)
add_executable(quality_eval eval.cpp ${root}/quality_core.cpp)
target_include_directories(quality_eval PRIVATE ${root} ${root}/engine/include ${root}/engine/ggml/include)
target_link_libraries(quality_eval PRIVATE llama)
